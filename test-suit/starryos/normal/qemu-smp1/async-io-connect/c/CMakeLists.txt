cmake_minimum_required(VERSION 3.20)
project(async-io-connect C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(server src/server.c)
target_compile_options(server PRIVATE -Wall -Wextra -Werror)
install(TARGETS server RUNTIME DESTINATION usr/bin)
add_executable(client src/client.c)
target_compile_options(client PRIVATE -Wall -Wextra -Werror)
install(TARGETS client RUNTIME DESTINATION usr/bin)
