cmake_minimum_required(VERSION 3.20)
project(crawl-mt C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(crawl-mt src/crawl-mt.c)
target_compile_options(crawl-mt PRIVATE -Wall -Wextra -Werror)
target_link_libraries(crawl-mt PRIVATE pthread)
install(TARGETS crawl-mt RUNTIME DESTINATION usr/bin)
